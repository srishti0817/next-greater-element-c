#include <stdio.h>

void printNGE(int arr[], int n) {
    int stack[100], top = -1;
    int nge[100];

    for (int i = n - 1; i >= 0; i--) {
        while (top != -1 && stack[top] <= arr[i]) {
            top--;
        }
        nge[i] = (top == -1) ? -1 : stack[top];
        stack[++top] = arr[i];
    }

    for (int i = 0; i < n; i++) {
        printf("%d -> %d\n", arr[i], nge[i]);
    }
}
